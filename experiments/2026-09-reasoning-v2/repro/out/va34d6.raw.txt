  { id: 2, date: "2023-10-22", entity: "Cloud Services", amount: 45000, type: "OpEx", status: "Pending" },

  { id: 6, date: "2023-10-20", entity: "Client B", amount: 150000, type: "Revenue", status: "Confirmed" },

const meetingAgenda = [
  { id: 1, time: "09:00 AM", topic: "Board Approval", status: "Ready" },
  { id: 2, time: "10:30 AM", topic: "Q3 Review", status: "In Progress" },
  { id: 3, time: "11:00 AM", topic: "Cash Flow Analysis", status: "Pending" },

export default function FinanceDashboard() {
  return (
    <Page title="Weekly Finance Pulse">

      <div style={{ display: "flex", flexDirection: "column", height: "100vh" }}>

          <Section title="Upcoming Agenda">
